// LAZY-LOAD: Carrega Mercado Pago APENAS quando necessário
// ============================================================

/**
 * Carrega script de terceiros sob demanda
 * @param {string} scriptId - ID único do script
 * @param {string} scriptSrc - URL do script a carregar
 * @param {function} callback - Função executada após carregar
 */
function czkCarregarScriptTerceiro(scriptId, scriptSrc, callback = null) {
    // Se já foi carregado, não carrega novamente
    if (document.getElementById(scriptId)) {
        if (callback) callback();
        return;
    }

    const script = document.createElement('script');
    script.id = scriptId;
    script.src = scriptSrc;
    script.async = true;

    if (callback) {
        script.onload = callback;
    }

    document.head.appendChild(script);
    console.log(`✅ Carregado: ${scriptId}`);
}

// ============================================================
// MERCADO PAGO: Carrega apenas ao clicar em "Pedir"
// ============================================================

function czkCarregarMercadoPago() {
    czkCarregarScriptTerceiro(
        'mercadopago-sdk',
        'https://sdk.mercadopago.com/js/v2',
        function() {
            // Inicializa Mercado Pago após carregar
            if (typeof MercadoPago !== 'undefined') {
                MercadoPago.Wallet({
                    initialization: {
                        amount: 100.00, // Substitua pelo valor real
                        payer: {
                            email: 'cliente@email.com' // Substitua dinamicamente
                        }
                    },
                    customization: {
                        visual: {
                            buttonColor: 'blue'
                        }
                    },
                    onReady: () => {
                        console.log('🟢 Mercado Pago pronto!');
                    },
                    onError: (error) => {
                        console.error('🔴 Erro ao carregar Mercado Pago:', error);
                    }
                });
            }
        }
    );
}

// Vincula ao clique do botão "Pedir"
document.addEventListener('DOMContentLoaded', function() {
    // Procura por todos os botões "Pedir"
    const botoesPedir = document.querySelectorAll('[data-acao="pedir"]');
    
    botoesPedir.forEach(btn => {
        btn.addEventListener('click', function(e) {
            // Carrega Mercado Pago na primeira vez que clica em "Pedir"
            czkCarregarMercadoPago();
        });
    });
});

// FONTAWESOME: Carrega apenas em painéis específicos
// ============================================================

function czkCarregarFontAwesome() {
    czkCarregarScriptTerceiro(
        'fontawesome-css',
        'https://cdnjs.cloudflare.com/ajax/libs/font-awesome/6.4.0/css/all.min.css'
    );
}

// Carrega FontAwesome quando abre painel Admin/Entregador/Cozinha
document.addEventListener('DOMContentLoaded', function() {
    const painelAdmin = document.getElementById('admin-panel');
    const painelEntregador = document.getElementById('entregador-panel');
    const painelCozinha = document.getElementById('cozinha-panel');

    function abrirPainel() {
        if (painelAdmin?.style.display === 'flex' || 
            painelEntregador?.style.display === 'flex' || 
            painelCozinha?.style.display === 'flex') {
            czkCarregarFontAwesome();
        }
    }

    // Observer para quando painéis abrem
    const observer = new MutationObserver(abrirPainel);
    observer.observe(document.body, { 
        attributes: true, 
        subtree: true,
        attributeFilter: ['style', 'class']
    });
});

// LEAFLET (Mapa): Carrega apenas no modal de entrega
// ============================================================

function czkCarregarLeaflet() {
    // CSS do Leaflet
    czkCarregarScriptTerceiro(
        'leaflet-css',
        'https://cdn.jsdelivr.net/npm/leaflet@1.9.4/dist/leaflet.min.css'
    );

    // JS do Leaflet
    czkCarregarScriptTerceiro(
        'leaflet-js',
        'https://cdn.jsdelivr.net/npm/leaflet@1.9.4/dist/leaflet.min.js',
        function() {
            console.log('🗺️ Leaflet carregado!');
        }
    );
}

// Carrega Leaflet quando abre modal de entrega
document.addEventListener('DOMContentLoaded', function() {
    const modalMapa = document.getElementById('modal-mapa-entrega');
    
    if (modalMapa) {
        const observer = new MutationObserver(() => {
            if (modalMapa.style.display === 'flex' || !modalMapa.classList.contains('hidden')) {
                czkCarregarLeaflet();
            }
        });

        observer.observe(modalMapa, { 
            attributes: true,
            attributeFilter: ['style', 'class']
        });
    }
});

// ONE SIGNAL (Notificações): Carrega apenas no app do cliente
// ============================================================

function czkCarregarOneSignal() {
    czkCarregarScriptTerceiro(
        'onesignal-sdk',
        'https://cdn.onesignal.com/sdks/OneSignalSDK.js',
        function() {
            window.OneSignal = window.OneSignal || [];
            OneSignal.push(function() {
                OneSignal.init({
                    appId: 'SEU_ONESIGNAL_APP_ID_AQUI',
                    allowLocalhostAsSecureOrigin: true
                });
            });
            console.log('🔔 OneSignal carregado!');
        }
    );
}

// Carrega OneSignal apenas quando cliente faz login (não na tela de login)
document.addEventListener('DOMContentLoaded', function() {
    const appCliente = document.getElementById('app-cliente');
    
    if (appCliente && appCliente.style.display !== 'none') {
        czkCarregarOneSignal();
    }
});
